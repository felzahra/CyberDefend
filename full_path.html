1. Cisco Core SwitchRouting to Palo Alto

CISCO IOS CLI 
# Default route pointing to Palo Alto Internal Firewall
ip route 0.0.0.0 0.0.0.0 10.10.10.2 # PA Trust IP

# Interface connecting to Palo Alto
interface GigabitEthernet0/1
 description TO-PALO-ALTO-TRUST
 no switchport
 ip address 10.10.10.1 255.255.255.252
 no shutdown

2. Palo Alto FirewallInternal Firewall Configuration

PALO ALTO CLI 
# Trust Interface (Connected to Core)
set network interface ethernet ethernet1/1 layer3 ip 10.10.10.2/30
set zone Trust network layer3 ethernet1/1

# Untrust Interface (Connected to Fortinet)
set network interface ethernet ethernet1/2 layer3 ip 192.168.1.2/30
set zone Untrust network layer3 ethernet1/2

# Security Policy: Allow internal users to go out
set rulebase security rules Allow-Outbound
    from Trust
    to Untrust
    source any
    destination any
    application [ web-browsing ssl dns ]
    service application-default
    action allow

# NAT: Source NAT to Fortinet IP
set rulebase nat rules Outbound-NAT
    from Trust
    to Untrust
    source-translation dynamic-ip-and-port
        translated-address interface-address
            interface ethernet1/2

# Static Route to Fortinet
set network virtual-router default routing-table ip static-route To-Fortinet destination 0.0.0.0/0 nexthop ip-address 192.168.1.1

3. Fortinet FirewallEdge Firewall Configuration

FORTIOS CLI 
# Interface connected to Palo Alto (Internal)
config system interface
    edit port1
        set ip 192.168.1.1 255.255.255.252
        set alias TO-PALO-ALTO
    next

# Interface connected to Edge Router (External)
    edit port2
        set ip 203.0.113.2 255.255.255.252
        set alias TO-ROUTER
    next
end

# Firewall Policy: Allow traffic from PA to Router
config firewall policy
    edit 1
        set srcintf port1
        set dstintf port2
        set srcaddr all
        set dstaddr all
        set action accept
        set schedule always
        set service ALL
        set nat enable # Source NAT to Router IP
    next
end

# Static Route to Edge Router
config router static
    edit 1
        set device port2
        set dst 0.0.0.0 0.0.0.0
        set gateway 203.0.113.1
    next
end

4. Edge RouterISP Connection

CISCO IOS CLI 
# Interface connected to Fortinet
interface GigabitEthernet0/0
 description TO-FORTINET-EXTERNAL
 ip address 203.0.113.1 255.255.255.252
 no shutdown

# Interface connected to ISP (Internet)
interface Serial0/0/0
 description TO-ISP-WAN
 ip address 8.8.8.2 255.255.255.252 # Example ISP IP
 no shutdown

# Default route to ISP
ip route 0.0.0.0 0.0.0.0 8.8.8.1 # ISP Gateway

# Route back to internal Fortinet subnet (if NAT is not used on FW)
ip route 192.168.1.0 255.255.255.252 203.0.113.2
